1. The Three Fundamental Problems in Habit Trackers

1. Shallow State

Standard trackers store a single boolean per day, discarding duration, difficulty, notes, or verification evidence.

2. Flawed Streaks

Hardcoded counters penalize users for scheduled rest days, causing burnout and streak fatigue.

3. Fragmented Tools

Habit tracking, deep work focus timers, and learning hubs are split across three separate subscriptions.

3. Deterministic Streak Engine Algorithm

HabitFlow calculates streaks mathematically from chronological database records. It introduces Rest Day Protection, allowing planned recovery without breaking streaks: 

// packages/shared/src/streakEngine.ts
export function calculateStreak(
  targetDate: string,
  historyMap: Record<string, 'COMPLETED' | 'REST' | 'MISSED' | 'SKIPPED'>
): StreakResult {
  let currentStreak = 0;
  let cursor = new Date(targetDate);

  while (true) {
    const dateStr = cursor.toISOString().split('T')[0];
    const status = historyMap[dateStr];

    if (status === 'COMPLETED') {
      currentStreak++;
    } else if (status === 'REST') {
      // Rest Day preserves streak without incrementing
      continue;
    } else {
      break;
    }
    cursor.setDate(cursor.getDate() - 1);
  }
  return { currentStreak, isActive: currentStreak > 0 };
}
